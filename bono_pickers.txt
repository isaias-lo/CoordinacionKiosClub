Mínimo estadístico de auditorías minimo = CEIL((Z² × 0.95 × 0.05) / e²) Con valores default (Z=1.96, e=0.05): minimo = CEIL((3.8416 × 0.0475) / 0.0025) = 73 pallets Este número sale de la tabla parametros_sistema y se recalcula automáticamente si el encargado cambia Z o e.B. Efectividad por picker (mes actual) efectividad = (auditorias_ok / auditorias_totales) × 100 donde:  auditorias_ok     = COUNT auditorías sin error del picker en el mes  auditorias_totales = COUNT total de auditorías del picker en el mes C. Cobertura de auditoría cobertura_picker = (auditados_picker / producidos_picker) × 100 cobertura_global = (total_auditados_dia / total_producidos_dia) × 100 D. CONSULTAS SQL PARA SUPABASE-- Vista: métricas mensuales por picker CREATE OR REPLACE VIEW picker_metricas_mes AS WITH params AS (  SELECT    MAX(CASE WHEN clave = 'minimo_auditorias' THEN valor END) AS minimo,    MAX(CASE WHEN clave = 'umbral_bono_pct'   THEN valor END) AS umbral  FROM parametros_sistema ), base AS (  SELECT    a.picker_nombre,    DATE_TRUNC('month', a.fecha::date)            AS mes,    COUNT(*)                                       AS auditados,    COUNT(*) FILTER (WHERE a.tiene_errores = false                     OR a.tiene_errores IS NULL)   AS ok,    COUNT(*) FILTER (WHERE a.tiene_errores = true) AS errores  FROM auditorias a   -- ← ajustar al nombre real de la tabla  WHERE a.picker_nombre IS NOT NULL  GROUP BY a.picker_nombre, DATE_TRUNC('month', a.fecha::date) ) SELECT  b.picker_nombre,  b.mes,  b.auditados,  b.ok,  b.errores,  ROUND((b.ok::numeric / NULLIF(b.auditados, 0)) * 100, 2)  AS efectividad_pct,  GREATEST(0, params.minimo::int - b.auditados)              AS deficit,  CASE    WHEN b.auditados >= params.minimo     AND ROUND((b.ok::numeric / b.auditados) * 100, 2) >= params.umbral      THEN 'BONO_APROBADO'    WHEN b.auditados >= params.minimo      THEN 'BONO_NEGADO'    ELSE 'PENDIENTE'  END AS estado_bono FROM base, params;-- Función: lista de prioridad diaria CREATE OR REPLACE FUNCTION prioridad_auditoria_hoy() RETURNS TABLE (  picker_nombre   text,  auditados_mes   int,  deficit         int,  necesarios_hoy  int ) AS $$ DECLARE  dias_restantes int; BEGIN  SELECT COUNT(*) INTO dias_restantes  FROM generate_series(    CURRENT_DATE + 1,    DATE_TRUNC('month', CURRENT_DATE) + INTERVAL '1 month - 1 day',    INTERVAL '1 day'  ) d  WHERE EXTRACT(DOW FROM d) BETWEEN 1 AND 5;  RETURN QUERY  SELECT    v.picker_nombre,    v.auditados::int,    v.deficit::int,    CASE      WHEN dias_restantes = 0 THEN v.deficit::int      ELSE CEIL(v.deficit::numeric / NULLIF(dias_restantes, 0))::int    END AS necesarios_hoy  FROM picker_metricas_mes v  WHERE v.mes = DATE_TRUNC('month', CURRENT_DATE)  ORDER BY necesarios_hoy DESC; END; $$ LANGUAGE plpgsql; IMPORTANTE: Los nombres de tabla y campos en el SQL de arriba ( auditorias , picker_nombre , tiene_errores , fecha ) son referencias.
